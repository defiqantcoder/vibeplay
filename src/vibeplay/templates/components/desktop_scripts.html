Theme Engine & Persistence function initThemeEngine() { const savedTheme = localStorage.getItem('vibeplay_theme') || 'god-of-ink'; applyTheme(savedTheme, false); } function applyTheme(themeId, persist = true) { document.documentElement.setAttribute('data-theme', themeId); document.body.setAttribute('data-theme', themeId); if (persist) { try { localStorage.setItem('vibeplay_theme', themeId); } catch (e) { console.warn('Could not persist theme to localStorage:', e); } } // Highlight active theme card in settings dialog document.querySelectorAll('[data-theme-card]').forEach(card => { if (card.getAttribute('data-theme-card') === themeId) { card.classList.add('active'); } else { card.classList.remove('active'); } }); // Close menus closeAllMenus(); } // 2. Menu Dropdown Interactions let activeOpenMenu = null; function toggleMenu(menuName) { const dropdown = document.getElementById(`dropdown-${menuName}`); if (!dropdown) return; if (activeOpenMenu === menuName) { closeAllMenus(); } else { closeAllMenus(); dropdown.classList.add('open'); activeOpenMenu = menuName; } } function closeAllMenus() { document.querySelectorAll('.menu-dropdown.open').forEach(el => { el.classList.remove('open'); }); activeOpenMenu = null; } // Close menus when clicking outside document.addEventListener('click', (e) => { if (!e.target.closest('.menu-item-wrapper')) { closeAllMenus(); } }); // 3. Settings Popover & Dialog Controls function toggleSettingsDialog(initialTab = null) { const dialog = document.getElementById('settings-dialog'); if (!dialog) return; if (dialog.open) { dialog.close(); } else { if (initialTab) { switchSettingsTab(initialTab); } if (typeof dialog.showModal === 'function') { dialog.showModal(); } else { dialog.setAttribute('open', ''); } } closeAllMenus(); } function closeSettingsDialog() { const dialog = document.getElementById('settings-dialog'); if (dialog && dialog.open) { dialog.close(); } } function switchSettingsTab(tabId) { const tabs = ['transport', 'dsp', 'analyzer', 'theme']; tabs.forEach(t => { const btn = document.getElementById(`tab-btn-${t}`); const panel = document.getElementById(`panel-${t}`); if (btn) { if (t === tabId) btn.classList.add('active'); else btn.classList.remove('active'); } if (panel) { if (t === tabId) panel.classList.add('active'); else panel.classList.remove('active'); } }); } // 4. Global Desktop Keyboard Shortcuts window.addEventListener('keydown', (e) => { const activeTag = document.activeElement ? document.activeElement.tagName.toLowerCase() : ''; const isInputActive = ['input', 'textarea', 'select'].includes(activeTag); triggerAddFolder(); return; } // Ctrl+,: Toggle Settings Popover if ((e.ctrlKey || e.metaKey) && e.key === ',') { e.preventDefault(); toggleSettingsDialog(); return; } // Ctrl+Q: Exit Vibeplay if ((e.ctrlKey || e.metaKey) && (e.key === 'q' || e.key === 'Q')) { e.preventDefault(); handleDesktopQuit(); return; } // F12 or Ctrl+F11: Toggle Fullscreen if (e.key === 'F12' || (e.ctrlKey && e.key === 'F11')) { e.preventDefault(); toggleFullscreen(); return; } // Volume Down: F5 or AudioVolumeDown if (e.key === 'F5' || e.code === 'F5' || e.key === 'AudioVolumeDown') { e.preventDefault(); adjustVolume(-5); return; } // Volume Mute: F6 or AudioVolumeMute if (e.key === 'F6' || e.code === 'F6' || e.key === 'AudioVolumeMute') { e.preventDefault(); triggerMuteToggle(); return; } // Volume Up: F7 or AudioVolumeUp if (e.key === 'F7' || e.code === 'F7' || e.key === 'AudioVolumeUp') { e.preventDefault(); adjustVolume(5); return; } // Previous Track: F9, MediaTrackPrevious, or Ctrl+ArrowLeft if (e.key === 'F9' || e.code === 'F9' || e.key === 'MediaTrackPrevious' || (e.ctrlKey && e.key === 'ArrowLeft')) { e.preventDefault(); triggerPrevTrack(); return; } // Play / Pause: F10, MediaPlayPause, or Space (when not in text input) if (e.key === 'F10' || e.code === 'F10' || e.key === 'MediaPlayPause' || (e.code === 'Space' && !isInputActive)) { e.preventDefault(); triggerPlayToggle(); return; } // Next Track: F11, MediaTrackNext, or Ctrl+ArrowRight if (e.key === 'F11' || e.code === 'F11' || e.key === 'MediaTrackNext' || (e.ctrlKey && e.key === 'ArrowRight')) { e.preventDefault();
